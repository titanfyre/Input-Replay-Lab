cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(InputReplayLab VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/SDK" CACHE PATH "Interceptor Keyboard & Mouse SDK directory; override privately for the production edition")
option(ENABLE_ASAN "Enable AddressSanitizer for offline validation" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /PDBALTPATH:%_PDB%)
if(ENABLE_ASAN)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(RecordingCore STATIC App/Recording.cpp)
target_include_directories(RecordingCore PUBLIC App)
add_library(InterceptorSDK STATIC "${SDK_DIRECTORY}/Interceptor.cpp")
target_include_directories(InterceptorSDK PUBLIC "${SDK_DIRECTORY}")
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_executable(InputReplayLab WIN32 App/Main.cpp App/Engine.cpp)
target_link_libraries(InputReplayLab PRIVATE RecordingCore InterceptorSDK comctl32 comdlg32 shell32 advapi32 dwmapi gdiplus)
target_link_options(InputReplayLab PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/App/App.manifest")
enable_testing()
add_executable(RecordingTests Tests/RecordingTests.cpp)
target_link_libraries(RecordingTests PRIVATE RecordingCore)
add_test(NAME RecordingModel COMMAND RecordingTests)
add_test(NAME OfflineApplication COMMAND InputReplayLab --self-test)
add_executable(EngineTests Tests/EngineTests.cpp App/Engine.cpp)
target_include_directories(EngineTests BEFORE PRIVATE Tests/Fake)
target_link_libraries(EngineTests PRIVATE RecordingCore)
add_test(NAME FakeTransportLifecycle COMMAND EngineTests)
